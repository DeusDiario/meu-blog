🚌 Objetivo

Desenvolver um sistema de monitoramento em tempo real da linha 8000 de ônibus da cidade de São Paulo, utilizando a API SPTrans e visualizando os dados em um mapa interativo.

🎯 Metodologia

O projeto utiliza:

  • API SPTrans: Dados em tempo real dos ônibus
  • Folium: Biblioteca para criação de mapas interativos
  • Requests: Para comunicação com a API

💻 Implementação

1. Classe para Comunicação com a API

Primeiro, criamos uma classe para gerenciar a autenticação e requisições:

import requests

class SPTransAPI:
    """
    Classe para interagir com a API Olho Vivo da SPTrans
    """
    
    def __init__(self, token):
        """
        Inicializa a conexão com a API
        
        Parâmetros:
        -----------
        token : str
            Token de autenticação da API SPTrans
        """
        self.token = token
        self.base_url = "http://api.olhovivo.sptrans.com.br/v2.1"
        self.authenticated = False
        self.session = requests.Session()
        
    def autenticar(self):
        """
        Autentica na API usando o token fornecido
        
        Retorna:
        --------
        bool : True se autenticado com sucesso, False caso contrário
        """
        url = f"{self.base_url}/Login/Autenticar?token={self.token}"
        
        try:
            response = self.session.post(url, data='')
            
            if response.status_code == 200 and response.text.strip().lower() == 'true':
                self.authenticated = True
                print("✅ Autenticado com sucesso!")
                return True
            else:
                print("❌ Falha na autenticação")
                return False
                
        except Exception as e:
            print(f"❌ Erro ao autenticar: {e}")
            return False

2. Busca de Linhas

Implementamos métodos para buscar informações sobre as linhas:

def buscar_linhas(self, termo):
    """
    Busca linhas de ônibus por termo
    
    Parâmetros:
    -----------
    termo : str
        Termo de busca (número ou nome da linha)
    
    Retorna:
    --------
    list : Lista de dicionários com informações das linhas
    """
    if not self.authenticated:
        print("❌ Você precisa autenticar primeiro!")
        return []
    
    url = f"{self.base_url}/Linha/Buscar?termosBusca={termo}"
    
    try:
        response = self.session.get(url)
        
        if response.status_code == 200:
            linhas = response.json()
            print(f"✅ Encontradas {len(linhas)} linhas")
            return linhas
        else:
            return []
            
    except Exception as e:
        print(f"❌ Erro: {e}")
        return []

def obter_posicoes(self, codigo_linha):
    """
    Obtém posições em tempo real dos ônibus
    
    Parâmetros:
    -----------
    codigo_linha : int
        Código da linha

        return []

5. Criação do Mapa Interativo

Usamos Folium para criar um mapa com as paradas e ônibus:

import folium
from folium import plugins

def criar_mapa_linha(api, codigo_linha, nome_linha):
    """
    Cria um mapa interativo com paradas e ônibus
    
    Parâmetros:
    -----------
    api : SPTransAPI
        Instância autenticada da API
    codigo_linha : int
        Código da linha
    nome_linha : str
        Nome descritivo da linha
    
    Retorna:
    --------
    folium.Map : Mapa interativo
    """
    # Obtém dados
    paradas = api.obter_paradas(codigo_linha)
    veiculos = api.obter_posicoes(codigo_linha)
    
    # Filtra paradas válidas
    paradas_validas = [p for p in paradas 
                       if p.get('py', 0) != 0 and p.get('px', 0) != 0]
    
    # Calcula centro do mapa
    lats = [p['py'] for p in paradas_validas]
    lons = [p['px'] for p in paradas_validas]
    centro = [sum(lats)/len(lats), sum(lons)/len(lons)]
    
    # Cria o mapa
    mapa = folium.Map(
        location=centro,
        zoom_start=13,
        tiles='OpenStreetMap'
    )
    
    # Adiciona paradas (marcadores azuis)
    grupo_paradas = folium.FeatureGroup(name='🚏 Paradas')
    
    for parada in paradas_validas:
        folium.Marker(
            location=[parada['py'], parada['px']],
            popup=folium.Popup(
                f"<b>{parada['np']}</b><br>"
                f"Código: {parada['cp']}<br>"
                f"Endereço: {parada['ed']}",
                max_width=300
            ),
            tooltip=f"🚏 {parada['np']}",
            icon=folium.Icon(color='blue', icon='stop', prefix='fa')
        ).add_to(grupo_paradas)
    
    grupo_paradas.add_to(mapa)
    
    # Adiciona ônibus (marcadores vermelhos)
    if veiculos:
        grupo_veiculos = folium.FeatureGroup(name='🚌 Ônibus')
        
        for veiculo in veiculos:
            if veiculo.get('py', 0) != 0 and veiculo.get('px', 0) != 0:
                folium.Marker(
                    location=[veiculo['py'], veiculo['px']],
                    popup=f"🚌 Ônibus {veiculo.get('p', 'N/A')}",
                    tooltip=f"🚌 {veiculo.get('p', 'N/A')}",
                    icon=folium.Icon(color='red', icon='bus', prefix='fa')
                ).add_to(grupo_veiculos)
        
        grupo_veiculos.add_to(mapa)
    
    # Adiciona controle de camadas
    folium.LayerControl().add_to(mapa)
    
    # Adiciona botão de tela cheia
    plugins.Fullscreen().add_to(mapa)
    
    return mapa

Usamos Folium para criar um mapa com as paradas e ônibus:

```python
import folium
from folium import plugins

def criar_mapa_linha(api, codigo_linha, nome_linha):
    """
    Cria um mapa interativo com paradas e ônibus
    
    Parâmetros:
    -----------
    api : SPTransAPI
        Instância autenticada da API
    codigo_linha : int
        Código da linha
    nome_linha : str
        Nome descritivo da linha
    
    Retorna:
    --------
    folium.Map : Mapa interativo
    """
    # Obtém dados
    paradas = api.obter_paradas(codigo_linha)
    veiculos = api.obter_posicoes(codigo_linha)
    
    # Filtra paradas válidas
    paradas_validas = [p for p in paradas 
                       if p.get('py', 0) != 0 and p.get('px', 0) != 0]
    
    # Calcula centro do mapa
    lats = [p['py'] for p in paradas_validas]
    lons = [p['px'] for p in paradas_validas]
    centro = [sum(lats)/len(lats), sum(lons)/len(lons)]
    
    # Cria o mapa
    mapa = folium.Map(
        location=centro,
        zoom_start=13,
        tiles='OpenStreetMap'
    )
    
    # Adiciona paradas (marcadores azuis)
    grupo_paradas = folium.FeatureGroup(name='🚏 Paradas')
    
    for parada in paradas_validas:
        folium.Marker(
            location=[parada['py'], parada['px']],
            popup=folium.Popup(
                f"<b>{parada['np']}</b><br>"
                f"Código: {parada['cp']}<br>"
                f"Endereço: {parada['ed']}",
                max_width=300
            ),
            tooltip=f"🚏 {parada['np']}",
            icon=folium.Icon(color='blue', icon='stop', prefix='fa')
        ).add_to(grupo_paradas)
    
    grupo_paradas.add_to(mapa)
    
    # Adiciona ônibus (marcadores vermelhos)
    if veiculos:
        grupo_veiculos = folium.FeatureGroup(name='🚌 Ônibus')
        
        for veiculo in veiculos:
            if veiculo.get('py', 0) != 0 and veiculo.get('px', 0) != 0:
                folium.Marker(
                    location=[veiculo['py'], veiculo['px']],
                    popup=f"🚌 Ônibus {veiculo.get('p', 'N/A')}",
                    tooltip=f"🚌 {veiculo.get('p', 'N/A')}",
                    icon=folium.Icon(color='red', icon='bus', prefix='fa')
                ).add_to(grupo_veiculos)
        
        grupo_veiculos.add_to(mapa)
    
    # Adiciona controle de camadas
    folium.LayerControl().add_to(mapa)
    
    # Adiciona botão de tela cheia
    plugins.Fullscreen().add_to(mapa)
    
    return mapa
```

import requests
import folium
from folium import plugins
import json

class SPTransAPI:
    
    def __init__(self, token):
        self.token = token
        self.base_url = "http://api.olhovivo.sptrans.com.br/v2.1"
        self.authenticated = False
        self.session = requests.Session()
        
    def autenticar(self):
        url = f"{self.base_url}/Login/Autenticar?token={self.token}"
        
        try:
            response = self.session.post(url, data='')
            
            print(f"🔡 Autenticando na API SPTrans...")
            print(f"   Status: {response.status_code}")
            print(f"   Resposta: {response.text}")
            
            if response.status_code == 200:
                result = response.text.strip().lower()
                if result == 'true':
                    self.authenticated = True
                    print("✅ Autenticado com sucesso!\n")
                    return True
                else:
                    print("❌ Autenticação falhou - token inválido\n")
                    return False
            else:
                print(f"❌ Erro HTTP: {response.status_code}\n")
                return False
                
        except Exception as e:
            print(f"❌ Erro ao autenticar: {e}\n")
            return False
    
    def buscar_linhas(self, termo):
        if not self.authenticated:
            print("❌ Você precisa autenticar primeiro!")
            return []
        
        url = f"{self.base_url}/Linha/Buscar?termosBusca={termo}"
        
        try:
            response = self.session.get(url)
            
            print(f"🔍 Buscando linhas com termo: '{termo}'")
            print(f"   Status: {response.status_code}")
            
            if response.status_code == 200:
                linhas = response.json()
                print(f"✅ Encontradas {len(linhas)} linhas\n")
                return linhas
            else:
                print(f"❌ Erro: {response.status_code}\n")
                return []
                
        except Exception as e:
            print(f"❌ Erro: {e}\n")
            return []
    
    def obter_paradas(self, codigo_linha):
        if not self.authenticated:

            return []
        
        url = f"{self.base_url}/Parada/BuscarParadasPorLinha?codigoLinha={codigo_linha}"
